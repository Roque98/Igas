Uso en Angular

Ejemplo en Componente

<!-- En tu template de Angular -->
<div class="card-igas">
  <div class="card-header-igas">
    {{ ticket.folio }}
  </div>
  <div class="card-body-igas">
    <p>{{ ticket.descripcion }}</p>

    <div class="component-row">
      <span
        class="badge-priority"
        [ngClass]="{
          'badge-priority-critical': ticket.prioridad === 'Crítica',
          'badge-priority-high': ticket.prioridad === 'Alta',
          'badge-priority-medium': ticket.prioridad === 'Media',
          'badge-priority-low': ticket.prioridad === 'Baja'
        }">
        {{ ticket.prioridad }}
      </span>

      <span
        class="badge-sla"
        [ngClass]="{
          'badge-sla-ok': ticket.porcentajeSLA < 70,
          'badge-sla-warning': ticket.porcentajeSLA >= 70 && ticket.porcentajeSLA <= 100,
          'badge-sla-critical': ticket.porcentajeSLA > 100
        }">
        SLA: {{ ticket.porcentajeSLA }}%
      </span>
    </div>
  </div>
</div>
